        function toggleFullScreen() {
            if (!document.fullscreenElement && !document.webkitFullscreenElement) {
                const docEl = document.documentElement;
                const requestFn = docEl.requestFullscreen || docEl.webkitRequestFullscreen;
                if (requestFn) {
                    requestFn.call(docEl).then(() => {
                        isUITemporarilyHidden = true;
                        applyUIHiddenState(true);
                    }).catch(err => {
                        console.warn('Fullscreen request denied', err);
                        isUITemporarilyHidden = true;
                        applyUIHiddenState(true);
                    });
                } else {
                    isUITemporarilyHidden = true;
                    applyUIHiddenState(true);
                }
            } else {
                exitFullScreenAndRestore();

        function checkUrlHashImport() {
            try {
                const hash = window.location.hash;
                if (hash && hash.startsWith('#data=')) {
                    const base64Data = hash.substring(6);
                    const jsonStr = decodeURIComponent(escape(atob(base64Data)));
                    const imported = JSON.parse(jsonStr);
                    if (Array.isArray(imported) && imported.length > 0) {
                        spots = imported;
                        saveData(true);
                        showToast('共有リンクからスポットデータを読み込みました！');
                    }
                }
            } catch (e) {
                console.error(e);
            }
        }

        async function saveCloudSyncSettings() {
            let url = document.getElementById('firebase-db-url').value.trim();
            if (!url) {
                disconnectCloudSync();
                closeSyncModal();
                return;
            }
            url = url.replace(/\/+$/, '');
            if (!url.startsWith('https://')) {
                showToast('https:// から始まるFirebaseのURLを入力してください', 'error');
                return;
            }
            cloudDbUrl = url;
            localStorage.setItem(CLOUD_URL_KEY, cloudDbUrl);
            updateSyncStatusUI();
            closeSyncModal();
            showToast('クラウド同期に接続しています...', 'info');
            await pullFromCloud(true);
            startSyncPolling();
        }

        function disconnectCloudSync() {
            cloudDbUrl = '';
            localStorage.removeItem(CLOUD_URL_KEY);
            if (syncInterval) clearInterval(syncInterval);
            updateSyncStatusUI();
            closeSyncModal();
            showToast('クラウド共有を解除し、端末内保存に切り替えました', 'info');
        }

        async function pushToCloud() {
            if (!cloudDbUrl) return;
            try {
                const payload = JSON.stringify(spots);
                lastSyncedHash = payload;
                const res = await fetch(`${cloudDbUrl}/spots.json`, {
                    method: 'PUT',
                    headers: { 'Content-Type': 'application/json' },
                    body: payload
                });
                if (!res.ok) throw new Error('Cloud write failed');
            } catch (e) {
                console.error('Cloud write error:', e);
            }
        }

        async function pullFromCloud(isInitial = false) {
            if (!cloudDbUrl) return;
            try {
                const res = await fetch(`${cloudDbUrl}/spots.json`);
                if (!res.ok) throw new Error('Cloud read failed');
                const data = await res.json();
                if (data && Array.isArray(data)) {
                    const incomingHash = JSON.stringify(data);
                    if (incomingHash !== lastSyncedHash) {
                        spots = data;
                        lastSyncedHash = incomingHash;
                        localStorage.setItem(STORAGE_KEY, JSON.stringify(spots));
                        renderApp();
                        if (!isInitial) {
                            showToast('他のユーザーの最新の変更を自動反映しました', 'info');
                        } else {
                            showToast('クラウドから最新の危険スポットを読み込みました');
                        }
                    }
                } else if (data === null && isInitial) {
                    if (spots.length > 0) {
                        await pushToCloud();
                    } else {
                        loadData();
                        await pushToCloud();
                    }
                }
            } catch (e) {
                if (isInitial) {
                    console.warn('Cloud sync error:', e);
                }
            }
        }

        function getFilteredSpots() {
            const lvlFilter = document.getElementById('filter-level').value;
            const searchQuery = document.getElementById('filter-search').value.trim().toLowerCase();

            return spots.filter(spot => {
                const matchLvl = lvlFilter === 'all' || spot.level === lvlFilter;
                const matchSearch = !searchQuery || 
                    spot.title.toLowerCase().includes(searchQuery) || 
                    (spot.memo && spot.memo.toLowerCase().includes(searchQuery));
                return matchLvl && matchSearch;
            });
        }

        function handlePhotoSelect(event) {
            const file = event.target.files[0];
            if (!file) return;

            if (!file.type.startsWith('image/')) {
                showToast('画像ファイルを選択してください', 'error');
                return;
            }

            const reader = new FileReader();
            reader.onload = function(e) {
                const img = new Image();
                img.onload = function() {
                    const canvas = document.createElement('canvas');
                    const MAX_WIDTH = 800;
                    const MAX_HEIGHT = 600;
                    let width = img.width;
                    let height = img.height;

                    if (width > height) {
                        if (width > MAX_WIDTH) {
                            height *= MAX_WIDTH / width;
                            width = MAX_WIDTH;
                        }
                    } else {
                        if (height > MAX_HEIGHT) {
                            width *= MAX_HEIGHT / height;
                            height = MAX_HEIGHT;
                        }
                    }

                    canvas.width = width;
                    canvas.height = height;
                    const ctx = canvas.getContext('2d');
                    ctx.drawImage(img, 0, 0, width, height);

                    const compressedBase64 = canvas.toDataURL('image/jpeg', 0.75);
                    setSpotPhotoPreview(compressedBase64);
                };
                img.src = e.target.result;
            };
            reader.readAsDataURL(file);
        }

        function saveSpot(e) {
            e.preventDefault();
            
            const id = document.getElementById('spot-id').value;
            const lat = parseFloat(document.getElementById('spot-lat').value);
            const lng = parseFloat(document.getElementById('spot-lng').value);
            const title = document.getElementById('spot-title').value.trim();
            const level = document.querySelector('input[name="danger-level"]:checked').value;
            const memo = document.getElementById('spot-memo').value.trim();
            const photo = document.getElementById('spot-photo-base64').value || null;

            const now = new Date();
            const updatedAt = `${now.getFullYear()}-${String(now.getMonth()+1).padStart(2,'0')}-${String(now.getDate()).padStart(2,'0')} ${String(now.getHours()).padStart(2,'0')}:${String(now.getMinutes()).padStart(2,'0')}`;

            const saved = localStorage.getItem(STORAGE_KEY);
            if (saved) {
                try {
                    spots = JSON.parse(saved);
                } catch (err) {
                    console.error(err);
                }
            }

            if (id) {
                const idx = spots.findIndex(s => s.id === id);
                if (idx !== -1) {
                    spots[idx] = { ...spots[idx], title, level, memo, photo, updatedAt };
                }
                showToast('危険スポットを更新しました');
            } else {
                const newSpot = {
                    id: 'spot-' + Date.now(),
                    title,
                    level,
                    memo,
                    photo,
                    lat,
                    lng,
                    updatedAt
                };
                spots.unshift(newSpot);
                showToast('新しい危険スポットを登録しました');
            }

            localStorage.setItem(STORAGE_KEY, JSON.stringify(spots));
            
            if (cloudDbUrl) {
                pushToCloud();
            }

            renderApp();
            closeSpotModal();

            const drawer = document.getElementById('floating-drawer');
            if (drawer) {
                drawer.style.height = '';
            }
        }

            const file = event.target.files[0];
            if (!file) return;

            const reader = new FileReader();

                        spots = imported;

                } catch (err) {

            };
            reader.readAsText(file);
            event.target.value = '';
        }

        function escapeHTML(str) {
            if (!str) return '';
            return str.replace(/[&<>'"]/g, tag => ({
                '&': '&amp;',
                '<': '&lt;',
                '>': '&gt;',
                "'": '&#39;',
